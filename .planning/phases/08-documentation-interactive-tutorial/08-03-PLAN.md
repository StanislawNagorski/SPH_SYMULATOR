`def check_step(step_n: int, line: str, last_sim_result: Optional[dict], strategies_keys: Set[str], builtin_strategies: frozenset, tutorial_flow: Optional['TutorialFlow'] = None) -> bool:`

Body: an explicit `if step_n == N: ...` dispatch matching the RESEARCH §Step Verification Map exactly. Pseudocode (translate to Python):

```
line = (line or '').strip()
tokens = line.split()

if step_n == 1:
    return (
        len(tokens) >= 2 and tokens[0] == 'run' and 'naive' in tokens
        and last_sim_result is not None
        and last_sim_result.get('avg_val_last100', 0) >= 80.0
    )

if step_n == 2:
    return line == 'strategies' or line.startswith('strategy ')

if step_n == 3:
    return (
        len(tokens) >= 2 and tokens[0] == 'run'
        and tokens[1] in builtin_strategies
        and last_sim_result is not None
        and last_sim_result.get('avg_val_last100', None) is not None
    )

if step_n == 4:
    # custom command — detection by STRATEGIES diff, not by line shape only
    return bool(set(strategies_keys) - set(builtin_strategies))

if step_n == 5:
    return (
        tokens and tokens[0] == 'compare'
        and last_sim_result is not None
        and 'comparison' in last_sim_result
        and last_sim_result['comparison'].get('delta')
    )

if step_n == 6:
    # Open Question #2 resolution — side-shell CLI demo, check for new reports dir
    if tutorial_flow is None:
        return False
    return bool(tutorial_flow.new_reports_dirs())

if step_n == 7:
    # Open Question #3 resolution — soft step, any non-empty line passes
    return bool(line)

if step_n == 8:
    return (
        tokens and tokens[0] == 'batch' and '--seeds' in line
        and last_sim_result is not None
        and 'aggregate' in last_sim_result
    )
```
