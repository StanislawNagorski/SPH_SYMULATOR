    if eval "$cmd" > /tmp/p7_check.log 2>&1; then
        echo "[PASS] $desc"; PASS=$((PASS + 1))
    else
        echo "[FAIL] $desc"; echo "       cmd: $cmd"; echo "       log:"
        sed 's/^/         /' /tmp/p7_check.log | head -20; FAIL=$((FAIL + 1))
    fi
}
