    Stworzyć `scripts/verify_phase1.sh` (bash, set -euo pipefail). Skrypt składa się z 6 sekcji, każda drukuje `[SC#] ...` na początku i `OK` / fail message na końcu:

    ```bash
    #!/usr/bin/env bash
    # verify_phase1.sh — pełna weryfikacja ROADMAP Success Criteria Phase 1
    # Uruchamiać z project root: ./scripts/verify_phase1.sh
    set -euo pipefail

    cd "$(dirname "$0")/.."

    echo "=== Phase 1: Refactoring foundation — verification ==="

    # SC#1 + SC#4: regression check (8 inwokacji bit-identyczne z fixtures)
    echo
    echo "[SC#1+4] Regression check (8 fixtures bit-identical)..."
    python scripts/regression_check.py
    echo "  OK — wszystkie 8 inwokacji produkują identyczne JSON"

    # SC#2: module line count (każdy ≤ 150 linii)
    echo
    echo "[SC#2] Module line counts (≤ 150 linii each)..."
    fail=0
    while IFS= read -r f; do
      n=$(wc -l < "$f" | tr -d ' ')
      if [ "$n" -gt 150 ]; then
        echo "  FAIL: $f = $n linii (limit 150)"
        fail=1
      else
        printf "  %-40s %4d linii\n" "$f" "$n"
      fi
    done < <(find sphsim -name '*.py' -type f | sort)
    if [ "$fail" -ne 0 ]; then
      echo "FAIL SC#2: są moduły powyżej 150 linii"
      exit 2
    fi
    echo "  OK — wszystkie moduły ≤ 150 linii"

    # SC#3: sph_sim.py uruchamialny jako entry point
    echo
    echo "[SC#3] python sph_sim.py jako entry point..."
    out=$(python sph_sim.py --strategy naive --seed 42 --json 2>&1)
    if ! echo "$out" | python -c "import sys, json; d=json.loads(sys.stdin.read()); assert d['strategy']=='naive'"; then
      echo "FAIL SC#3: sph_sim.py nie działa jako entry point"
      exit 3
    fi
    echo "  OK — python sph_sim.py działa i produkuje JSON"

    # D-06: python -m sphsim też działa
    echo
    echo "[D-06] python -m sphsim alternatywny entry point..."
    out=$(python -m sphsim --strategy naive --seed 42 --json 2>&1)
    if ! echo "$out" | python -c "import sys, json; d=json.loads(sys.stdin.read()); assert d['strategy']=='naive'"; then
      echo "FAIL D-06: python -m sphsim nie działa"
      exit 4
    fi
    echo "  OK — python -m sphsim działa"

    # D-16: publiczne API
    echo
    echo "[D-16] Publiczne API: from sphsim import SPHSimulator, Device, STRATEGIES..."
    python -c "
    from sphsim import SPHSimulator, Device, STRATEGIES
    assert callable(SPHSimulator)
    assert callable(Device) or hasattr(Device, '__dataclass_fields__')
    assert isinstance(STRATEGIES, dict)
    assert set(STRATEGIES.keys()) >= {'naive', 'threshold', 'phase_prob', 'incentive', 'adaptive'}
    "
    echo "  OK — wszystkie 3 nazwy publicznego API importowalne"

    # D-07: BEZ pyproject.toml / setup.cfg
    echo
    echo "[D-07] Negative constraint: brak pyproject.toml / setup.cfg..."
    if [ -f pyproject.toml ] || [ -f setup.cfg ] || [ -f setup.py ]; then
      echo "FAIL D-07: znaleziono pyproject.toml/setup.cfg/setup.py — niedozwolone w Phase 1"
      ls -la pyproject.toml setup.cfg setup.py 2>/dev/null || true
      exit 7
    fi
    echo "  OK — projekt pozostaje 'lokalny, nie publikowany'"

    # Phase 1 constraint: stdlib only
    echo
    echo "[Phase 1 constraint] stdlib only — brak nowych zależności..."
    if ! grep -E "^\s*(import|from)\s+" sphsim/**/*.py sphsim/*.py 2>/dev/null \
         | grep -vE "(import|from)\s+(argparse|json|random|math|dataclasses|typing|sphsim|os|sys|subprocess|pathlib)" \
         | grep -v "^$"; then
      echo "  OK — pakiet sphsim używa tylko stdlib + własnych modułów"
    else
      echo "  WARN: znaleziono potencjalne non-stdlib importy (sprawdź manualnie):"
      grep -E "^\s*(import|from)\s+" sphsim/**/*.py sphsim/*.py 2>/dev/null \
        | grep -vE "(import|from)\s+(argparse|json|random|math|dataclasses|typing|sphsim|os|sys|subprocess|pathlib)" || true
    fi

    # Summary
    echo
    echo "=== ALL CHECKS PASSED — Phase 1 spełnia wszystkie ROADMAP Success Criteria ==="
    echo "Można rozpocząć Phase 2 (Interactive CLI shell)."
    ```
